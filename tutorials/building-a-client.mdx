An MTConnect client is any application that consumes data from an Agent via HTTP. Here's a complete Python client.

```python client.py
import requests
import xml.etree.ElementTree as ET

BASE = "http://localhost:5000"

def get_current():
    r = requests.get(f"{BASE}/current")
    root = ET.fromstring(r.text)
    # Observations are returned as type-specific elements (e.g. <Execution>,
    # <ControllerMode>, <PathPosition>) rather than a generic <DataItem> tag —
    # see /concepts/streams-and-samples. Walk every element instead and use
    # the dataItemId attribute, which every observation carries, to identify
    # them.
    return {
        item.get("dataItemId"): item.text
        for item in root.iter()
        if item.get("dataItemId")
    }

if __name__ == "__main__":
    data = get_current()
    for key, val in data.items():
        print(f"  {key}: {val}")
```
